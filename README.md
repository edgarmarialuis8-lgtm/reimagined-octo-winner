@Composable
fun App() {
    var started by remember { mutableStateOf(false) }
    if (!started) Cover { started = true } else Setup()
}

@Composable
fun Cover(onStart: () -> Unit) {
    Surface(Modifier.fillMaxSize()) {
        Column(
            Modifier.fillMaxSize().padding(28.dp),
            verticalArrangement = Arrangement.Center
        ) {
            Text("🎮", style = MaterialTheme.typography.displayLarge)

            Spacer(Modifier.height(32.dp))
            Button(onClick = onStart, Modifier.fillMaxWidth()) { Text("COMEÇAR") }
            Spacer(Modifier.height(12.dp))
            Text("v1.0 • Android")
        }
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun Setup() {
    var phone by remember { mutableStateOf(phones.first()) }
    var style by remember { mutableStateOf("Equilibrado") }
    var weapon by remember { mutableStateOf("Geral") }
    var generated by remember { mutableStateOf(false) }
    var seed by remember { mutableIntStateOf(1) }

    LazyColumn(
        Modifier.fillMaxSize().padding(20.dp),
        verticalArrangement = Arrangement.spacedBy(14.dp)
    ) {
        item {
            Text("Criar recomendação", style = MaterialTheme.typography.headlineSmall)
            Text("Escolhe o teu aparelho e o teu estilo de jogo.")
        }
        item { Selector("📱 Aparelho", phone, phones) { phone = it } }
        item { Selector("🎯 Estilo", style, listOf("Rush","Equilibrado","Precisão","Sniper")) { style = it } }
        item { Selector("🔫 Arma", weapon, listOf("Geral","SMG","AR","Shotgun","Sniper")) { weapon = it } }
        item {
            Button({ seed = Random.nextInt(); generated = true }, Modifier.fillMaxWidth()) {
                Text("GERAR SENSIBILIDADE")
            }
        }
        if (generated) item {
            val base = (when(style){"Rush"->92;"Precisão"->82;"Sniper"->70;else->87} + seed % 5).coerceIn(0,100)
            Card(Modifier.fillMaxWidth()) {
                Column(Modifier.padding(18.dp)) {
                    Text("🎯 Resultado", style = MaterialTheme.typography.titleLarge, fontWeight = FontWeight.Bold)
                    Text("Aparelho: $phone")
                    Text("Perfil: $style • $weapon")
                    Spacer(Modifier.height(10.dp))
                    Line("Geral", base)
                    Line("Ponto Vermelho", base-4)
                    Line("Mira 2×", base-9)
                    Line("Mira 4×", base-14)
                    Line("Mira AWM", base-30)
                    Line("Olhadinha", base-12)
                    Spacer(Modifier.height(10.dp))
                    Text("Estes valores são um ponto de partida. Ajusta-os durante o treino conforme a resposta da mira.")
                }
            }
        }
    }
}

@Composable
fun Line(name:String, value:Int) {
    Row(Modifier.fillMaxWidth(), horizontalArrangement=Arrangement.SpaceBetween) {
        Text(name); Text(value.coerceIn(0,100).toString(), fontWeight=FontWeight.Bold)
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun Selector(title:String, value:String, options:List<String>, onPick:(String)->Unit) {
    var open by remember { mutableStateOf(false) }
    ExposedDropdownMenuBox(open, { open = !open }) {
        OutlinedTextField(value, {}, readOnly=true, label={Text(title)},
            modifier=Modifier.fillMaxWidth().menuAnchor())
        ExposedDropdownMenu(open, {open=false}) {
            options.forEach { DropdownMenuItem(text={Text(it)}, onClick={onPick(it);open=false}) }
        }
    }
}
